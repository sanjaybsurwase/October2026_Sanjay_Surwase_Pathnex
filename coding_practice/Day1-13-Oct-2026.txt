#### Install Nginx on pathnex server
---
- name: Install Nginx on pathnex server
  hosts: all
  become: yes

  tasks:
    - name: Install nginx
      yum:
        name: nginx
        state: present



#### Creating EC2 instance
provider "aws"{region = "ap-south-1"}
resource "aws_instance" "PathnexEC2" {ami = "ami-0cd4165aeb4c1a32d" instance_type = "c5.xlarge"
tags = {Name = "Pathnex-EC2" }}

##### Kubernetes Task — Create Nginx Pod
apiVersion: v1
kind: Pod
metadata:
  name: pathnex-pod
spec:
  containers:
    - name: web
      image: nginx:latest
      ports:
        - containerPort: 80
